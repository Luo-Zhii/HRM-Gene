        const itemYearMonth = `${itemDate.getFullYear()}-${String(itemDate.getMonth() + 1).padStart(2, '0')}`;
        return itemYearMonth === month;
      });
    }
    if (type && type !== 'All') {
      list = list.filter(item => item.type.toLowerCase() === type.toLowerCase());
    }
 
    return list;
  }
